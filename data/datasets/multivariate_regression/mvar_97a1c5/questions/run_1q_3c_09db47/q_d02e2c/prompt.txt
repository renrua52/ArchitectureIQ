## Dataset
Multivariate regression on `[0, 1]^n`: input is a vector `x ∈ R^n`, target is a scalar `y ∈ R` from a fixed symbolic expression. Train and test splits are fixed before any candidate training.

### Synthesis (PyTorch)
```python
def target(x: torch.Tensor) -> torch.Tensor:
    return torch.cos(2 * torch.pi * x[:, 0]) * x[:, 1] + (x[:, 0] - 2) ** 2 - x[:, 1]


def synthesize(
    *,
    train_size: int = 256,
    test_size: int = 256,
    point_seed: int = 5212,
    input_dim: int = 2,
    domain_low: float = 0.0,
    domain_high: float = 1.0,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    gen = torch.Generator().manual_seed(point_seed)
    train_x = torch.rand(train_size, input_dim, generator=gen) * (domain_high - domain_low) + domain_low
    test_x = torch.rand(test_size, input_dim, generator=gen) * (domain_high - domain_low) + domain_low
    train_y = target(train_x)
    test_y = target(test_x)
    return train_x, train_y.unsqueeze(-1), test_x, test_y.unsqueeze(-1)
```

    """One hidden layer: leaky_relu(Linear(norm(x)))."""

    def __init__(self, width: int, use_layer_norm: bool) -> None:
        super().__init__()
        self.norm = nn.LayerNorm(width) if use_layer_norm else nn.Identity()
        self.linear = nn.Linear(width, width)
        self.act = nn.LeakyReLU(0.01)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        h = self.linear(self.norm(x))
        return self.act(h)

            MLPBlock(width=96, use_layer_norm=True),
            nn.Linear(96, 1),
